let battleState=null;
function startBattle(enemy){battleState={enemy,player:team[activeIndex]};renderBattle()}
function renderBattle(){const e=battleState.enemy,p=battleState.player;openModal(`<div class="modal-head"><h2>⚔️ COMBATE</h2><button class="close" onclick="closeModal()">✕</button></div><div class="battle-layout"><div class="fighter"><b>${e.name} · Lv.${e.level}</b><img src="${e.sprite}"><div class="hp"><i style="width:${e.hp/e.maxHP*100}%"></i></div><small>❤️ ${e.hp}/${e.maxHP}</small></div><div class="fighter"><b>${p.name} · Lv.${p.level}</b><img src="${p.sprite}"><div class="hp"><i style="width:${p.hp/p.maxHP*100}%"></i></div><small>❤️ ${p.hp}/${p.maxHP}</small></div></div><div class="actions" style="margin-top:14px">${p.moves.map(m=>`<button onclick="attack('${m}')">${MOVES[m].name}</button>`).join('')}<button onclick="tryCapture()">🎯 Capturar</button><button onclick="closeModal()">🏃 Huir</button></div>`)}
function attack(moveId){const e=battleState.enemy,p=battleState.player,m=MOVES[moveId];let dmg=Math.max(1,Math.floor((p.attack+ p.level)*m.power/Math.max(25,e.defense)*(.85+Math.random()*.15)));e.hp=Math.max(0,e.hp-dmg);if(e.hp<=0){alert(`¡${e.name} fue derrotado!`);coins+=50;save();closeModal();return}let em=e.moves[0]||'tackle',mm=MOVES[em];let ed=Math.max(1,Math.floor((e.attack+e.level)*mm.power/Math.max(25,p.defense)*(.85+Math.random()*.15)));p.hp=Math.max(0,p.hp-ed);if(p.hp<=0){p.hp=p.maxHP;alert(`${p.name} se debilitó y volvió al centro Pokémon.`);healTeam();}save();renderBattle();renderTeam()}
function tryCapture(){if(team.length>=6){alert('Tu equipo está lleno.');return}if(Math.random()>.58){alert('¡El Pokémon escapó de la Poké Ball!');return}const p=battleState.enemy;p.hp=p.maxHP;team.push(p);dexSeen.add(p.id);alert(`¡${p.name} fue capturado!`);save();closeModal();renderTeam()}
